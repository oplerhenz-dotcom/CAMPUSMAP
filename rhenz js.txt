function focusCamera(building) {

    cameraAnimating = true;
    cameraProgress = 0;

    // Current camera position
    cameraStart.copy(camera.position);

    // Current OrbitControls target
    targetStart.copy(controls.target);

    // Building center
    const targetPosition = new THREE.Vector3(
        building.x,
        building.h / 2,
        building.z
    );

    targetEnd.copy(targetPosition);

    // Camera destination
    cameraEnd.set(
        building.x + 15,
        building.h + 12,
        building.z + 18
    );
}

function gate(x, z, label) {
    const group = new THREE.Group();

    const cream = new THREE.MeshStandardMaterial({
        color: 0xf5e6c8
    });

    const metal = new THREE.MeshStandardMaterial({
        color: 0x171717,
        metalness: 0.5,
        roughness: 0.5
    });

    // SIDE PILLARS
    for (const side of [-1, 1]) {
        const pillar = new THREE.Mesh(
            new THREE.BoxGeometry(0.8, 6, 0.8),
            cream
        );

        pillar.position.set(side * 5, 3, 0);
        group.add(pillar);
    }

    // ARCH TOP
    const points = [];

    for (let i = 0; i <= 20; i++) {
        const angle = Math.PI - (Math.PI * i / 20);

        points.push(new THREE.Vector3(
            5 * Math.cos(angle),
            3.5 + 3 * Math.sin(angle),
            0
        ));
    }

    const archCurve = new THREE.CatmullRomCurve3(points);

    const arch = new THREE.Mesh(
        new THREE.TubeGeometry(archCurve, 40, 0.4, 8, false),
        cream
    );

    group.add(arch);

    // VERTICAL METAL BARS
    for (let i = -4.3; i <= 4.3; i += 0.5) {
        const bar = new THREE.Mesh(
            new THREE.BoxGeometry(0.1, 3.5, 0.12),
            metal
        );

        bar.position.set(i, 1.75, 0);
        group.add(bar);
    }

    // HORIZONTAL METAL BARS
    for (const y of [0.8, 2.8]) {
        const rail = new THREE.Mesh(
            new THREE.BoxGeometry(9, 0.12, 0.12),
            metal
        );

        rail.position.set(0, y, 0);
        group.add(rail);
    }

    // GATE LABEL
    const s = labelCanvas(label);
    s.position.set(0, 7.5, 0);
    s.scale.set(5, 1.2, 1);

    group.add(s);

    group.position.set(x, 0, z);
    scene.add(group);
    
    group.rotation.y = Math.PI / 2;
}

function animate(){

    requestAnimationFrame(animate);

    // Update animation time
    markerTime += 0.03;

    // Floating pin animation
    pin.position.y = originalPinY + Math.sin(markerTime * 2) * 0.25;

    // Keep white center attached to the pin
    center.position.y = pin.position.y;

    // Floating label animation
    locationLabel.position.y = originalLabelY + Math.sin(markerTime * 2) * 0.25;

    // Pulsing ground circle
    const pulse = 1 + Math.sin(markerTime * 2) * 0.2;

    circle.scale.set(pulse, pulse, 1);

    circle.material.opacity = 0.25 + (Math.sin(markerTime * 2) + 1) * 0.15;
    
    // Smooth camera movement
if (cameraAnimating) {

    cameraProgress += 0.025;

    const t = Math.min(cameraProgress, 1);

    // Smooth easing
    const smooth = t * t * (3 - 2 * t);

    // Move camera
    camera.position.lerpVectors(
        cameraStart,
        cameraEnd,
        smooth
    );

    // Move camera focus
    controls.target.lerpVectors(
        targetStart,
        targetEnd,
        smooth
    );

    if (t >= 1) {
        cameraAnimating = false;
    }
}

    controls.update();
    renderer.render(scene,camera);
}

function roomsForFloor(b, key) {
  if (b.floors) {
    return b.floors[key] || [];
  }

  return key === "ground" ? b.rooms || [] : [];
}


function highlightBuilding(b){
  buildings.forEach(item=>{
    if(!item.mesh) return;
    const active=item===b;
    item.mesh.material.emissive.setHex(active?0xc00000:0x000000);
    item.mesh.material.emissiveIntensity=active?0.48:0;
    item.mesh.material.color.copy(item.originalColor);
  });
  
}
